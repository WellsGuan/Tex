\section{Main Results}

\subsection{The Approximate SPDE}

\begin{proposition}
    Let $\sigma \in \text{Lip}(\R, (0,\infty))$, and $\ell\in(2,4]$.There is a constant $C = C(\sigma,\ell) < \infty$ such the following holds. Let $v_t^{\rho}$ solve the following equation with $v_0^{\rho} \in \mathscr{X}_0^{\ell}$
    \[
    v_t^{\rho}(x) = \mathcal{G}_{t-s}v_s^{\rho}(x) + \gamma_{\rho}\int_s^t\mathcal{G}_{t+\rho-r}[\sigma_{\rho}(v_r^{\rho})dW_r](x).
    \]
    Define
    \[
    \sigma_{\rho}(x) := \gamma_{\rho}^{-1}\sigma(x),\quad x\in \R.
    \]
    For $T_0,T_1$ satisfying
    \[
    (TBA)
    \]
    we define for $t\geq T_0$,
    \[
    \widehat{B}_t^{\rho}(x) = \gamma_{\rho}\int_{T_0}^t \mathcal{G}_{T_1+\rho-r}[d\widetilde{W}_r^{\rho}](x)
    \]
    where $d\widetilde{W}^{\rho}$ satisfying
    \[
    \left\lVert \int_{T_0}^t\mathcal{G}_{\eta_r}[\sigma_{\rho}(v_r^{\rho})dW_r - (\mathcal{S}_{\zeta_r}\sigma_{\rho}^2(v_r^{\rho}))^{1/2}d\widetilde{W}_r]^{\rho}\right\rVert_2^2 \leq \dfrac{1}{2\pi^3}\int_{T_0}^t \zeta_r^2\eta_r^{-2}\lVert|\sigma_{\rho}(v_r^{\rho})|\rVert_2^2dr
    \]
    for $\zeta_r = [L(1/\rho)^{-1}(T_1-r)]^{1/2}$ and arbitrary postive path $(\eta_r)_{r\geq T_0}$, where
    \[
    \lVert|\sigma_{\rho}(v_r^{\rho})|\rVert_l^l :=\sup_{x\in\R^2}\mathbb{E}|\sigma_{\rho}(v_r^{\rho})(x)|^l.
    \]
    Then we have
    \[\begin{aligned}
\mathbb{E}
&\left|
V_t^{\rho,T_1}(x)
-
V_{T_0}^{\rho,T_1}(x)
-
\int_{T_0}^{t}
J_{\sigma_{\rho}}\left(
    S_{\rho}(T_1-r),
    V_r^{\rho,T_1}(x)
\right)
\,d\widehat{B}_r(x)
\right|^2 \\
\leq
&(TBA)
\end{aligned}
\]
where
\[
V_s^{\rho,t}(x):= \mathcal{G}_{t-s}v_s^{\rho}(x).
\]
\end{proposition}
\begin{proof}
    We may notice that, for all
\(t \in [T_0,T_1]\),
\begin{equation*}
V_t^{\rho,T_1}(x)
=
V_{T_0}^{\rho,T_1}(x)
+
\gamma_{\rho}\int_{T_0}^{t}
\mathcal{G}_{T_1+\rho-r}
\bigl[\sigma_{\rho}(v_r^{\rho})\,dW_r\bigr](x).
\end{equation*}
So that we know
\[
\begin{aligned}
    &\left(\mathbb{E}
\left|
V_t^{\rho,T_1}(x)
-
V_{T_0}^{\rho,T_1}(x)
-
\int_{T_0}^{t}
J_{\sigma_{\rho}}\left(
    S_{\rho}(T_1-r),
    V_r^{\rho,T_1}(x)
\right)
\,d\widehat{B}_r(x)
\right|^2\right)^{1/2} \\
&\leq 
\left(\mathbb{E}
\left|
\gamma_\rho
\int_{T_0}^{t}
\mathcal{G}_{T_1+\rho-r}
\left[
 \sigma_{\rho}(v_r^{\rho})\,dW_r
 -
 \left(\mathcal{S}_{\zeta_r}
       [\sigma_{\rho}^2\circ v_r^{\rho}]\right)^{1/2}
 d\widetilde W_r^{\rho}
\right](x)
\right|^2\right)^{1/2} + \\
&\qquad\left(\mathbb{E}\left|
\gamma_{\rho}\int_{T_0}^t \mathcal{G}_{T_1+\rho-r}\left[\left(\mathcal{S}_{\zeta_r}[\sigma_{\rho}^2\circ v_r^{\rho}]^{1/2}\right)d\widetilde{W}_r^{\rho}\right](x)\right.\right.\\ &\qquad- \left.\left.\int_{T_0}^t J_{\sigma_{\rho}}(S_{\rho}(T_1-r),V_r^{\rho,T_1}(x))d\widehat{B}_r(x) 
\right|^2\right)^{1/2}
\end{aligned}
\]
We choose
\(\eta_r := T_1+\rho-r\) to find a
space-time white noise \(d\widetilde W^{\rho}\) such that, for any
\(x\in\mathbb{R}^2\),
\begin{align*}
&\mathbb{E}
\left|
\gamma_\rho
\int_{T_0}^{t}
\mathcal{G}_{T_1+\rho-r}
\left[
 \sigma_{\rho}(v_r)\,dW_r
 -
 \left(\mathcal{S}_{\zeta_r}
       [\sigma_{\rho}^2\circ v_r^{\rho}]\right)^{1/2}
 d\widetilde W_r^{\rho}
\right](x)
\right|^2
\nonumber\\
&\qquad\le
\dfrac{1}{L(1/\rho)^2}\int_{T_0}^t\dfrac{T_1-r}{(T_1+\rho-r)^2}\lVert|\sigma_{\rho}(v_r^{\rho})|\rVert_2^2 dr \\
&\qquad\le
\dfrac{\lVert \sigma\rVert_{\infty}^2}{L(1/\rho)}\int_{T_0}^{T_1} \dfrac{T_1-r}{(T_1+\rho-r)^2}dr \\
&\qquad\le
\dfrac{\lVert \sigma\rVert_{\infty}^2}{L(1/\rho)}\left(L((T_1-T_0)/\rho) - 1 + \rho/(T_1-T_0 +\rho)\right)
\leq {C(\sigma)S_{\rho}(T_1-T_0)}
\end{align*}

Then, we compute
\[
\begin{aligned}
&\mathbb{E}\left|
\gamma_{\rho}\int_{T_0}^t \mathcal{G}_{T_1+\rho-r}\left[\left(\mathcal{S}_{\zeta_r}[\sigma_{\rho}^2\circ v_r^{\rho}]^{1/2}\right)d\widetilde{W}_r^{\rho}\right](x)\right.\\ &\qquad- \left.\int_{T_0}^t J_{\sigma_{\rho}}(S_{\rho}(T_1-r),V_r^{\rho,T_1}(x))d\widehat{B}_r(x) 
\right|^2 \\
= &\dfrac{4\pi}{L(1/\rho)}\mathbb{E}\left\lvert\int_{T_0}^t \int G_{T_1+\rho-r}(x-y)\left(\mathcal{S}_{\zeta_r}[\sigma_{\rho}^2\circ v_r^{\rho}]^{1/2}\right)d\widetilde{W}_r^{\rho}(y) \right. \\
&\qquad\left. -\int_{T_0}^t\int J_{\sigma_{\rho}}(S_{\rho}(T_1-r),V_r^{\rho,T_1}(x))G_{T_1+\rho-r}(x-y)d\widetilde{W}_r^{\rho}(y) 
\right\rvert^2 \\
\leq& \dfrac{4\pi}{L(1/\rho)}\int_{T_0}^t\int G_{T_1+\rho-r}(x-y)^2F(x,y)dydr
\end{aligned}
\]
where
\[
F(x,y):= \mathbb{E}\left|\left(\mathcal{S}_{\zeta_r}[\sigma_{\rho}^2\circ v_r^{\rho}]^{1/2}\right)(y) - J_{\sigma_{\rho}}(S_{\rho}(T_1-r),V_r^{\rho,T_1}(x))\right|^2
\]
\end{proof}

\subsection{First stage approximation}

\begin{proposition}
    Fix \(M,\lambda\in(0,\infty)\) and $T>0$, \(\ell\in(2,4]\). There is a constant
\[
C=C(M,\lambda,\ell,T)
\in(0,\infty)
\]
such that the following holds.
Suppose that \(\rho\in(0,C^{-1})\).
Let \(\sigma\) satisfy
\(\operatorname{Lip}(\sigma)\leq\lambda\), $|\sigma|_{\infty}^2 \leq M$, and let
\((v_t)_{t\geq 0}\) solve
\[
    v_t^{\rho}(x) = \mathcal{G}_{t-s}v_s^{\rho}(x) + \gamma_{\rho}\int_s^t\mathcal{G}_{t+\rho-r}[\sigma_{\rho}(v_r^{\rho})dW_r](x)
\]
with initial condition
\(v_0\in\mathscr{X}_0^\ell\).
Let
\(\xi\in[0,\sqrt{w(\rho)\rho}]\),
\begin{equation*}
t\in
\left[
 \left(
   [1+\mathrm{L}(1/\rho)]\kappa_{\ell,\rho}^{-1}
   +\kappa_{\ell,\rho}
 \right)\xi^2,\, T
\right],\quad \kappa_{\ell,\rho} = L(1/\rho)^{-1/(1-2/\ell)}
\end{equation*}
\textit{ for some $w:(0,C^{-1}) \to\R$ such that for any $M>0$,
\[\omega(\rho)L(1/\rho)^M \to 0,\quad \rho \to 0.\]
We let \(x,z\in\mathbb{R}^2\), and \(\iota_1,\iota_2\in[0,1]\).
Then we have}
\[
\begin{aligned}
\mathbb{E}&
\left|
 \left(
   \mathcal{S}_{\xi,z}
   [\sigma_{\rho}^2\circ v^{\rho}_t](x)
 \right)^{1/2}
 -
 J_{\sigma_{\rho},\rho}
 \left(
   S_{\rho}\!\left(
     \mathrm{L}(1/\rho)^{l_1}\xi^2
   \right),
   \mathcal{G}_{\mathrm{L}(1/\rho)^{l_2}\xi^2}
   v_t^{\rho}(x)
 \right)
\right|^{l_2} \\
&\quad\leq
C
\left\langle
 \lVert v_0\rVert_{\ell}
\right\rangle^2
 {L}\left({L}(1/\rho)\right).
\end{aligned}
\]
\end{proposition}
\begin{proof}
The proof combines the earlier results of this section with specific choices of
parameters. We proceed in several steps.

\medskip
\noindent\textit{Step 1: Choosing the parameters.}
Let
\begin{equation*}
\ttR=\ttB_{\xi,k_{\xi,z}(x)+z},\quad
\tau_0=t-(\kappa_{\ell,\rho}^{-1}+\kappa_{\ell,\rho})\xi^2,
\quad
\tau_1=\tau_0+\kappa_{\ell,\rho}^{-1}\xi^2,
\quad
\tau_2=\tau_1+\kappa_{\ell,\rho}\xi^2=t.
\end{equation*}
Let $X\in\RR^d$ denote the center of $\ttR$. Note that
\begin{equation*}
\tau_2\geq\tau_1\geq\tau_0
\geq\sfL(1/\rho)\kappa_{\ell,\rho}^{-1}\xi^2\geq0
\end{equation*}
by the assumption on $t$.\par
With the assumption of the upper bound of $t$, we know the following relations:
\[
\dfrac{\tau_2-\tau_1}{\tau_1-\tau_0} = \kappa_{\ell,\rho}^2,\quad \dfrac{\tau_2-\tau_1}{\rho} \leq {\kappa_{\ell,\rho}}\omega(\rho),\quad \dfrac{\tau_1-\tau_0}{\rho} \leq \kappa_{\ell,\rho}^{-1}\omega(\rho).
\]
and
\[
\Xi_{2;t-s}^{\sigma,\rho}[v_s]^2 \leq \left(\lVert v_s\rVert_2^2 + \dfrac{L(1/\rho)}{4\pi}(S_{\rho}(t-s))|\sigma|_{\infty}^2\right) \vee |\sigma|_{\infty}^2
\]
\medskip
\noindent\textit{Step 2: Flattening the solution.}
By the matrix-valued reverse triangle inequality and the flattening estimate,
there is an absolute constant
$C<\infty$ such that, as long as $\rho$ is small enough
\begin{align*}
E_1
&:= \EE\left\lvert
\left(\fint_{\ttR}\sigma_{\rho}^2(v^{\rho}_t(y))\,\dif y\right)^{1/2}
-\left(\fint_{\ttR} \sigma_{\rho}^2\!\left(
\clV_{\tau_1,\tau_2}\clZ_X\clG_{\tau_1-\tau_0}v^{\rho}_{\tau_0}(y)
\right)\,\dif y\right)^{1/2}
\right\rvert^{2}
\\
&\leq \fint_{\ttR}\EE\left\lvert
\sigma_{\rho}\!\left(\clV_{\tau_0,\tau_2}v^{\rho}_{\tau_0}(y)\right)
-\sigma_{\rho}\!\left(
\clV_{\tau_1,\tau_2}\clZ_X\clG_{\tau_1-\tau_0}v^{\rho}_{\tau_0}(y)
\right)
\right\rvert^{2}\,\dif y
\\&\leq \dfrac{\lambda^2L(1/\rho)}{4\pi}\fint_{\ttR}\EE\left\lvert
\clV_{\tau_0,\tau_2}v^{\rho}_{\tau_0}(y)
-\clV_{\tau_1,\tau_2}\clZ_X\clG_{\tau_1-\tau_0}v^{\rho}_{\tau_0}(y)
\right\rvert^{2}\,\dif y
\\
&\leq \dfrac{\lambda^2L(1/\rho)}{2\pi}\left(
\dfrac{ML\left(
\frac{\tau_1-\tau_0}
{t-\tau_1+\rho}
\right)}{4\pi - \lambda^2 L\left(\frac{\tau_2-\tau_1}{\rho}\right)}\right.
\\ &\left.\qquad +\|v_{\tau_0}\|_2^2\left(
\frac{
\tau_2-\tau_1+\frac12|X-x|^2
}{
\tau_1-\tau_0
}\right)
\left(\dfrac{4\pi}{4\pi - \lambda^2 L\left(\dfrac{\tau_2-\tau_1}{\rho}\right)}\right)
\right)
\\
&\leq CL(1/\rho)\left(L(\kappa_{\ell,\rho}^{-1}\omega(\rho)) + \kappa_{\ell,\rho}^2 + \kappa_{\ell,\rho}\right),
\end{align*}
for some contant $C_1(\sigma)$ and hence
\begin{align*}
E_1 \leq CL(1/\rho)L(\kappa_{\ell,\rho}^{-1}\omega(\rho)) .
\end{align*}
\medskip
\noindent\textit{Step 3: Concentration.}
Define
\[
E_2\coloneqq\EE\left\lvert
\left(\fint_{\ttR}\sigma_{\rho}^2\!\left(
\clV_{\tau_1,\tau_2}\clZ_X\clG_{\tau_1-\tau_0}v^{\rho}_{\tau_0}(y)
\right)\,\dif y\right)^{1/2}
-J_{\sigma_{\rho},\rho}\!\left(
\sfS_\rho(\tau_2-\tau_1),
\clG_{\tau_1-\tau_0}v^{\rho}_{\tau_0}(X)
\right)
\right\rvert_{\Frob}^{2}.
\]

and we have
\[
\begin{aligned}
E_2 &\leq 4\Xi_{2;\tau_2-\tau_1}[\clG_{\tau_1-\tau_0}v^{\rho}_{\tau_0}(X)]^2(\gamma_{\rho}^{-1}\lambda)^2S_{\rho}(\tau_2-\tau_1)  \\
& \leq 16\pi\left(\lVert v_{\tau_1}\rVert_2^2 + \dfrac{L(1/\rho)}{4\pi}(S_{\rho}(\tau_2-\tau_1))|\sigma|_{\infty}^2\right)\lambda^2 L\left(\dfrac{\tau_2-\tau_1}{\rho}\right) \\
&\leq CL(\kappa_{\ell,\rho}\omega(\rho))^2
\end{aligned}
\]

for some contant $C_2:= C_2(\sigma)$.

\medskip
\noindent\textit{Step 4: Adjusting the arguments of $J_{\sigma_{\rho},\rho}$.}
Let $\iota_1,\iota_2\in[0,1]$. By the regularity estimate for
$J_{\sigma_{\rho},\rho}$ and the triangle inequality
on $L^2(\Omega,\mathbb P)$,
\begin{align*}
&\left(\EE\left\lvert
J_{\sigma_{\rho},\rho}\!\left(
\sfS_\rho(\kappa_{\ell,\rho}\xi^2),
\clG_{\kappa_{\ell,\rho}^{-1}\xi^2}v_{\tau_0}(X)
\right)
-J_{\sigma_{\rho},\rho}\!\left(
\sfS_\rho(\sfL(1/\rho)^{\iota_1}\xi^2),
\clG_{\sfL(1/\rho)^{\iota_2}\xi^2}v_t(X)
\right)
\right\rvert^2\right)^{1/2}
\nonumber\\
&\quad \leq \left(\EE\left|J_{\sigma_{\rho},\rho}\!\left(
\sfS_\rho(\kappa_{\ell,\rho}\xi^2),
\clG_{\kappa_{\ell,\rho}^{-1}\xi^2}v_{\tau_0}(X)
\right) - J_{\sigma_{\rho},\rho}\!\left(
\sfS_\rho(\kappa_{\ell,\rho}\xi^2),
\clG_{\sfL(1/\rho)^{\iota_2}\xi^2}v_t(X)
\right)\right|^2 \right)^{1/2}
\\
&+ \left(\EE\left|J_{\sigma_{\rho},\rho}\!\left(
\sfS_\rho(\kappa_{\ell,\rho}\xi^2),
\clG_{\sfL(1/\rho)^{\iota_2}\xi^2}v_t(X)
\right)-J_{\sigma_{\rho},\rho}\!\left(
\sfS_\rho(\sfL(1/\rho)^{\iota_1}\xi^2),
\clG_{\sfL(1/\rho)^{\iota_2}\xi^2}v_t(X)\right)\right|^2 \right)^{1/2} \\
&\quad\leq
\left(\dfrac{L(1/\rho)\lambda^2}{4\pi}\EE\left|\clG_{\kappa_{\ell,\rho}^{-1}\xi^2}v_{\tau_0}(X) - \clG_{\sfL(1/\rho)^{\iota_2}\xi^2}v_t(X)\right|^2\right)^{1/2}
\nonumber\\
&\quad+\left(\dfrac{\lambda^2L(1/\rho)}{4\pi}\lVert\mathcal{V}_p\clG_{\sfL(1/\rho)^{\iota_2}\xi^2}v_t(X) - \mathcal{V}_{p-q,q}\clG_{\sfL(1/\rho)^{\iota_2}\xi^2}v_t(X)\rVert_2^2\right)^{1/2}\\
&\quad\leq \left(C\Xi_{2,t}[v_0]^2{L(L(1/\rho))}\right)^{1/2}+\left(CL(1/\rho)\dfrac{ML\left(
\frac{q}
{p-q+\rho}
\right)}{4\pi - \lambda^2 L\left(\frac{p-q}{\rho}\right)}\right)^{1/2} \\
&\quad \leq
\left(C\left(\lVert v_s\rVert_2^2 + \dfrac{L(1/\rho)}{4\pi}(S_{\rho}(t))|\sigma|_{\infty}^2\right){L(L(1/\rho))}\right)^{1/2}+\left(CL(1/\rho)L(\kappa_{\ell,\rho}w(\rho))\right)^{1/2}
\end{align*}
where
\[
p = (\kappa_{\ell,\rho}\xi^2 \vee L(1/\rho)^{\iota_1}\xi^2),\quad q = (\kappa_{\ell,\rho}\xi^2 \wedge L(1/\rho)^{\iota_1}\xi^2)
\]
by using the continuity estimate and
$\kappa_{\ell,\rho}^{-1}>\sfL(1/\rho)$, there is an absolute constant
$C<\infty$ such that, for sufficiently small $\rho$,
\begin{align*}
&
\EE\left\lvert
\clG_{\kappa_{\ell,\rho}^{-1}\xi^2}v_{\tau_0}(X)
-\clG_{\sfL(1/\rho)^{\iota_2}\xi^2}v_t(X)
\right\rvert^2
\nonumber\\
&\quad= \EE\left\lvert\clG_{\kappa_{\ell,\rho}^{-1}\xi^2}v_{\tau_0}(X)
-\clG_{\sfL(1/\rho)^{\iota_2}\xi^2}\left(\clG_{t-\tau_0}v_{\tau_0} + \int_{\tau_0}^t \clG_{t+\rho-r}[\sigma(v_r)dW_r]\right)(X)\right\rvert^2 \\
&\quad \leq \EE\left|\clG_{\kappa_{\ell,\rho}^{-1}\xi^2}v_{\tau_0}(X) - \clG_{\sfL(1/\rho)^{\iota_2}\xi^2 + t-\tau_0}v_{\tau_0}(X)\right|^2\\&\qquad + M\int_{\tau_0}^t \int G_{t+\sfL(1/\rho)^{\iota_2}\xi^2 +\rho -r}^2(X-y)dydr \\
&\leq \lVert v_{\tau_0}\rVert_2^2\dfrac{|(L(1/\rho)^{\iota_2}+\kappa_{\ell,\rho})\xi^2|}{\kappa_{\ell,\rho}^{-1}\xi^2}+ \dfrac{M}{4\pi}L\left(\dfrac{t-\tau_0}{\sfL(1/\rho)^{\iota_2}\xi^2 + \rho}\right)
\end{align*}
To sum up, we have
\[
E_3 \leq CL(1/\rho)L(L(1/\rho))
\]

\end{proof}
