\section{Mathematical Background and Project Proposal}

\subsection{Definitions and Modelling}

\begin{definition}
    [Stochastic Volatility Model]\label{def: sv-model}
Let $\{P_t\}_{t\geq 0}$ be a positive price time series, and define
the \emph{log return} by
\[
r_t=\log P_t-\log P_{t-1}, \qquad t\geq 1.
\]
We model the returns as
\[
r_t=\mu_r+\sigma_t Z_{2,t},
\]
where $\sigma_t>0$ is a latent random scale. Define the latent
log-variance, often called the \emph{latent log-volatility}, by
\[
h_t=\log(\sigma_t^2).
\]
The basic Gaussian stochastic volatility model is
\[
\begin{aligned}
h_t
&=\mu_h+\phi(h_{t-1}-\mu_h)+qZ_{1,t},\\
r_t
&=\mu_r+\exp(h_t/2)Z_{2,t},
\qquad t\geq 1,
\end{aligned}
\]
where $\mu_h,\mu_r\in\mathbb R$, $|\phi|<1$, and $q>0$.
The random variables
\[
\{Z_{i,t}:i\in\{1,2\},\ t\geq 1\}
\]
are mutually independent standard normal random variables and
are independent of $h_0$.\par
In particular,
\[
r_t\mid h_t
\sim\mathcal N(\mu_r,e^{h_t}),
\]
so $\sigma_t^2=e^{h_t}$ is the conditional variance of $r_t$
given $h_t$, rather than its unconditional variance.
\end{definition}
\vspace{1em}
\begin{proposition}
    Assume
\[
h_0\sim
\mathcal N\left(\mu_h,\frac{q^2}{1-\phi^2}\right),
\]
then $h_t$ will be a stationary process.
\end{proposition}
\begin{proof}
    To see that $h_t$ is stationary, we denote
    \(
    p_t := h_t - \mu_h
    \)
    and then
    \[
    p_t = \phi^tp_0 +\sum\limits_{k=1}^t q\phi^{t-k}Z_{1,k}.
    \]
    We have
    \[
    \begin{aligned}
    \text{Cov}(p_s,p_t) &= \phi^{t+s}\mathbb Ep_0^2 + \sum\limits_{k=1}^{t\wedge s} q^2\phi^{t+s-2k} \\
    &= \phi^{t+s}\left(\mathbb Ep_0^2 +q^2\dfrac{\phi^{2(t\wedge s)}-1}{1-\phi^2}\right) \\
    &=q^2\phi^{|t-s|}/(1-\phi^2)
    \end{aligned}
    \]
    if $p_0$ has the required distribution.
\end{proof}

\begin{theorem}[Optimal Affine Prediction]
\label{thm: optimal-linear-estimation}
Let $X\in\mathbb R^d$ and $Y\in\mathbb R^m$ have finite second
moments. Denote
\[
\begin{aligned}
    m_X&=\mathbb EX, & m_Y&=\mathbb EY,\\
    C_{XX}&=\mathbb E[(X-m_X)(X-m_X)^\top], &
    C_{YY}&=\mathbb E[(Y-m_Y)(Y-m_Y)^\top],\\
    C_{XY}&=\mathbb E[(X-m_X)(Y-m_Y)^\top], &
    C_{YX}&=C_{XY}^\top.
\end{aligned}
\]
Assume that $C_{YY}$ is positive definite, and set
\[
    K=C_{XY}C_{YY}^{-1},\qquad
    \ell(y)=m_X+K(y-m_Y).
\]
Then $\ell(Y)$ is the unique affine predictor minimizing
\[
    \mathbb E\|X-a-BY\|^2,
    \qquad a\in\mathbb R^d,\quad B\in\mathbb R^{d\times m}.
\]
The residual $U=X-\ell(Y)$ satisfies
\[
    \mathbb EU=0,\qquad
    \mathbb E[U(Y-m_Y)^\top]=0,
\]
and its covariance is
\[
    C_U=C_{XX}-C_{XY}C_{YY}^{-1}C_{YX}.
\]
In particular, the minimum mean squared error is
$\operatorname{tr}(C_U)$.
\end{theorem}

\begin{proof}
By the definition of $K$, we know
\[
    \mathbb EU=0,\qquad
    \mathbb E[U(Y-m_Y)^\top]=C_{XY}-KC_{YY}=0.
\]
For any $a$ and $B$, write
\[
    X-a-BY
    =U+(m_X-a-Bm_Y)+(K-B)(Y-m_Y).
\]
The three terms on the right have zero cross terms in the expected
squared norm, and hence
\[
\begin{aligned}
    \mathbb E\|X-a-BY\|^2
    &=\mathbb E\|U\|^2+\|m_X-a-Bm_Y\|^2\\
    &\quad+\operatorname{tr}\bigl((K-B)C_{YY}(K-B)^\top\bigr).
\end{aligned}
\]
Since $C_{YY}$ is positive definite, equality with the minimum holds
exactly when $B=K$ and $a=m_X-Km_Y$. Expanding
$\mathbb E[UU^\top]$ gives the stated covariance, and we are done.
\end{proof}

\begin{proposition}[Exact Posterior Update in the Gaussian Case]
\label{prop: gaussian-posterior-update}
Under the assumptions and notation of
Theorem~\ref{thm: optimal-linear-estimation}, assume further that
$(X,Y)$ is jointly Gaussian. Then a version of the conditional law is
\[
    \mathcal L(X\mid Y=y)
    =\mathcal N\bigl(\ell(y),C_U\bigr),
    \qquad y\in\mathbb R^m.
\]
Let $(X^f,Y^f)$ be an auxiliary pair with the same joint law as $(X,Y)$,
independent of the actual observation. For a fixed observed value $y$,
define
\[
    X^a=X^f+K(y-Y^f).
\]
Then
\[
    \mathcal L(X^a)=\mathcal L(X\mid Y=y).
\]
Thus the update gives both the conditional mean and the conditional
covariance, and in this case it gives the full posterior law.
\end{proposition}

\begin{proof}
The residual $U=X-m_X-K(Y-m_Y)$ and $Y$ are jointly Gaussian.
The previous theorem gives $\operatorname{Cov}(U,Y)=0$, so they are
independent. We also know that
\[
    X=\ell(Y)+U,\qquad U\sim\mathcal N(0,C_U).
\]
Conditioning on $Y=y$ therefore gives the stated Gaussian conditional
law. For the auxiliary pair, put
\[
    U^f=X^f-m_X-K(Y^f-m_Y).
\]
Then $U^f$ has the same law as $U$, and
\[
    X^a=m_X+K(y-m_Y)+U^f=\ell(y)+U^f.
\]
This has distribution $\mathcal N(\ell(y),C_U)$, and hence it is a draw
from the posterior law.
\end{proof}

The positive definiteness assumption above makes $C_{YY}^{-1}$
well defined; the residual covariance $C_U$ may still be singular.
Without joint Gaussianity, the first theorem still gives the best
affine predictor, but the residual need not be independent of $Y$.
The distributional conclusion in the proposition then need not hold.
Also, the proposition uses the population covariance matrices and an
exact joint forecast pair. Replacing these covariances by estimates
from a finite ensemble introduces an approximation.


\subsection{EnKF Algorithm and Pseudocode}

\begin{definition}[Abstraction of Ensemble Kalman Filtering]
    Consider the state-space equations
\[
\begin{aligned}
X_t &= \Phi(X_{t-1}) + W_t,
& W_t &\sim \mathcal N(0,Q_t),\\
Y_t &= H(X_t) + V_t,
& V_t &\sim \mathcal N(0,R_t),
\end{aligned}
\]
where $X_t\in\mathbb R^d$ is the latent state and
$Y_t\in\mathbb R^m$ is the observation. Assume that the noises are
mutually independent over time and independent of the initial state.
Denote the observed values by $y_{1},y_2,\cdots,y_t$ and we like to obtain the filtering distribution denoted as
\[
\pi_t=P(X_t\mid Y_1=y_1,\cdots,Y_t = y_t).
\]
\end{definition}

\begin{definition}
    [Ensemble Kalman Filtering for SV Model]
    Assume for the SV model, the parameters $(\mu_r,\mu_h,\phi,q)$ are given, then for the model:
    \[
    \begin{aligned}
        H_t &= \mu_h + \phi(H_{t-1}-\mu_h) + qZ_{1,t}, \\
        R_t &= \mu_r + \exp\left({H_t/2}\right)Z_{2,t}
    \end{aligned}
    \]
    where $Z_{i,t}$ are white noises defined as \ref{def: sv-model} and we use the captical $H,R$ for the time series in case it mess up with the observations involved later. Let $N$ denotes the ensemble size.\par
    We state the algorithm inductively, assume the current ensemble without updating \[h_{t-1}^{(1)},h_{t-1}^{(2)}\cdots,h_{t-1}^{(N)},\]
    which provide an approximation:
    \[
    \widehat{P}(H_{t-1}|R_1=r_1,\cdots,R_{t-1}=r_{t-1}) = \widehat{\pi}_{t-1}^N:= \dfrac{1}{N}\sum\limits_{i=1}^N \delta_{h_{t-1}^{(i)}}.
    \]
    Then we may use the evolution equation to obtain
    \[
    \widehat{P}(H_t|R_1 = r_1,\cdots,R_{t-1}=r_{t-1}) = \widetilde{\pi}_{t}^N:=\dfrac{1}{N}\sum\limits_{i=1}^N\delta_{\tilde{h}_t^{(i)}},
    \]
    where \[\tilde{h}_t^{(i)} = \mu_h + \phi(h_{t-1}^{(i)}-\mu_h)+qz_{1,t}^{(i)}\]
    for $z_{1,t}^{(i)}$ new standard normal realizations.\par
    After we obtained the latest observation $r_t$ pf $R_t$, we would like to use the \ref{thm: optimal-linear-estimation} to adjust the ensemble set by
    \[
    h_t^{(i)} := \tilde{h}_t^{(i)}+\dfrac{\widehat{\text{Cov}}(H,S)(t)}{\widehat{\text{Cov}}(S,S)(t)}(s_t-s_t^{(i)})
    \]
    where
    \[
    s_t:= (r_t-\mu_r)^2,\quad s_t^{(i)} :=\exp(\tilde{h}_t^{(i)})(z_{2,t}^{(i)})^2
    \]
    for $z_{2,t}^{(i)}$ new standard normal realizations, and
    \[
    \begin{aligned}
        \widehat{\text{Cov}}(H,S)(t)&:= \dfrac{1}{N-1}\sum\limits_{i=1}^N(\tilde{h}_t^{(i)}-\bar{h}_t)(s_t^{(i)}-\overline{s}_t),\\
        \widehat{\text{Cov}}(S,S)(t)&:= \dfrac{1}{N-1}\sum\limits_{i=1}^N(s_t^{(i)}-\overline{s}_t)(s_t^{(i)}-\overline{s}_t),
    \end{aligned}
    \]
    with
    \[\bar{h}_t:= \dfrac{1}{N}\sum\limits_{i=1}^N \tilde{h}_t^{(i)}, \quad
        \bar{s}_t:= \dfrac{1}{N}\sum\limits_{i=1}^N {s}_t^{(i)},\]
    then an approximation of filtering distribution for $h_t$ will be given as
    \[
    \widehat{\pi}_t^N: = \dfrac{1}{N}\sum\limits_{i=1}^N \delta_{h_t^{(i)}}.
    \]
\end{definition}

\begin{algorithm}[htbp]
\caption{EnKF-Type Filtering for the SV Model Using Squared Returns}
\label{alg:sv-enkf}
\begin{algorithmic}[1]
\Require Observed returns $r_{1:T}$; parameters
$(\mu_r,\mu_h,\phi,q)$ with $|\phi|<1$, $q>0$;
ensemble size $N\geq 2$.
\Ensure Approximate filtering distributions
$\widehat{\pi}_t^N$, $t=1,\ldots,T$.

\State Independently sample
$h_0^{(i)}\sim
\mathcal{N}\!\left(\mu_h,q^2/(1-\phi^2)\right)$,
$i=1,\ldots,N$.

\For{$t=1,\ldots,T$}
    \State $s_t\gets(r_t-\mu_r)^2$

    \For{$i=1,\ldots,N$}
        \State Draw independent standard normal realizations
        $z_{1,t}^{(i)}$ and $z_{2,t}^{(i)}$,
        independently of the existing ensemble.
        \State $\widetilde{h}_t^{(i)}
        \gets \mu_h+\phi(h_{t-1}^{(i)}-\mu_h)
        +qz_{1,t}^{(i)}$
        \State $s_t^{(i)}
        \gets \exp(\widetilde{h}_t^{(i)})
        \bigl(z_{2,t}^{(i)}\bigr)^2$
    \EndFor

    \State $\overline{h}_t
    \gets N^{-1}\sum_{i=1}^N\widetilde{h}_t^{(i)}$
    \State $\overline{s}_t
    \gets N^{-1}\sum_{i=1}^N s_t^{(i)}$

    \State $C_t^{hs}\gets
    \dfrac{1}{N-1}\sum_{i=1}^N
    \bigl(\widetilde{h}_t^{(i)}-\overline{h}_t\bigr)
    \bigl(s_t^{(i)}-\overline{s}_t\bigr)$

    \State $C_t^{ss}\gets
    \dfrac{1}{N-1}\sum_{i=1}^N
    \bigl(s_t^{(i)}-\overline{s}_t\bigr)^2$

    \If{$C_t^{ss}>0$}
        \State $K_t\gets C_t^{hs}/C_t^{ss}$
    \Else
        \State $K_t\gets 0$
        \Comment{Skip the update if the sample variance is zero}
    \EndIf

    \For{$i=1,\ldots,N$}
        \State $h_t^{(i)}
        \gets \widetilde{h}_t^{(i)}
        +K_t\bigl(s_t-s_t^{(i)}\bigr)$
    \EndFor

    \State $\widehat{\pi}_t^N
    \gets N^{-1}\sum_{i=1}^N\delta_{h_t^{(i)}}$
\EndFor

\State \Return $\{\widehat{\pi}_t^N\}_{t=1}^T$
\end{algorithmic}
\end{algorithm}
