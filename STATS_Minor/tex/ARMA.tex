\section{Stationary ARMA Processes}

\subsection{ARMA Processes}

\begin{definition}
    [White Noise] The process $(Z_t)$ is said to be white noise with mean $0$ and
variance $\sigma^2$ if and only if $(Z_t)$ has zero mean and covariance function
\[
\gamma(h) = \begin{cases}
    0,\quad &h \neq 0 \\ \sigma^2, & h = 0 
\end{cases}
\]
\end{definition}

\begin{definition}
    [The ARMA$(p, q)$ Process] The process  $(X_t)_{t\in\mathbb{Z}}$ is said to be an ARMA -$(p,q)$ process if $(X_t)$ is stationary and if for every $t$,
    \[
    X_t - \phi_1 X_{t-1} - \cdots \phi_p X_{t-p} = Z_t + \theta_1 Z_{t-1} + \cdots + \theta_q Z_{t-q}
    \]
where $(Z_t) \sim \text{WN}(0,\sigma^2)$. We say that $(X_t)$ is an ARMA$(p,q)$ process with mean $\mu$ if $(X_t - \mu)$ is an ARMA$(p,q)$ process.\par
The equation can be written as 
\[
\phi(B)X_t = \theta(B)Z_t
\]
where $\phi,\theta$ are $p,q$-degree polynomials and $B$ is the push-backward operator.
\end{definition}

\begin{definition}
    [The MA$(q)$ process and the AR$(p)$ process]
    The MA$(q)$ process is an ARMA$(0,q)$ process, that is $\phi \equiv 1$, i.e.
    \[
    X_t = \theta(B)Z_t.
    \]\par
    The AR$(p)$ process is an ARMA$(p,0)$ process, that is $\theta \equiv 1$, i.e.
    \[
    \phi(B)X_t = Z_t
    \]
\end{definition}

\begin{proposition}
    The mean value and covariance function for an MA$(q)$ process $X_t$ with MA polynomial $\theta$ is
    \[
    \mu_X(t) = 0,\quad \gamma_X(h) = \sigma^2\left(\sum\limits_{k=0}^{q-h}\theta_k\theta_{k+h}\right).
    \]
\end{proposition}
\begin{proof}
    We know
    \[
    \mathbb{E} X_t = \mathbb{E}\theta(B)Z_t = 0
    \]
    and
    \[
    \gamma_X(h) = \mathbb{E}\left(\mathbb{E}\theta(Z)Z_t\right)\left(\theta(Z)Z_{t+h}\right) = \sigma^2\left(\sum\limits_{k=0}^{q-h}\theta_k\theta_{k+h}\right)
    \]
    where $\theta_0:=1$.
\end{proof}

\begin{definition}[Causality]
An ARMA$(p,q)$ process defined by the equations
$\phi(B)X_t=\theta(B)Z_t$ is said to be \emph{causal}
(or more specifically to be a \emph{causal function of $\{Z_t\}$})
if there exists a sequence of constants $\{\psi_j\}_{j\geq0}$ such that
\[
\sum_{j=0}^{\infty}|\psi_j|<\infty
\]
and
\[
X_t=\sum_{j=0}^{\infty}\psi_j Z_{t-j},
\qquad t=0,\pm1,\ldots.
\]
\end{definition}

\begin{proposition}
If $\{X_t\}_{t\in\mathbb{Z}}$ is any sequence of random variables such that
$\sup_t\mathbb{E}|X_t|<\infty$, and if
$\sum_{j=-\infty}^{\infty}|\psi_j|<\infty$, then the series
\[
\psi(B)X_t
=\sum_{j=-\infty}^{\infty}\psi_j B^jX_t
=\sum_{j=-\infty}^{\infty}\psi_j X_{t-j}
\]
converges absolutely with probability one. If in addition
$\sup_t\mathbb{E}|X_t|^2<\infty$, then the series converges in mean square
to the same limit.
\end{proposition}
\begin{proof}
    To see the absolute convergence for a.s., we may notice that
    \[
    \mathbb{E} |\phi|(B)|X_t| \leq \sum\limits_{j=-\infty}^{\infty}|\phi_j|\sup\mathbb{E}|X_t| < \infty
    \]
    and hence $|\phi|(B)|X_t|$ has to be finite in probability 1.\par
    Then we would like to show that the series converge to $\phi(B)X_t$ in $L^2$ as well, that is we may prove that by noticing
    \[
    \mathbb{E}\left|\sum\limits_{m < |j| \leq n} \phi_j X_{t-j}\right|^2 \leq \left(\sum\limits_{|j|\geq m} |\phi_j|\right)^2\sup_{t}\mathbb{E}|X_t|^2 \to 0
    \]
    and hence the series converge in $L^2$ to some limit $L$, with
    \[
    \mathbb{E}|L-\sum\limits_{j = -\infty}^{\infty} \phi_jX_{t-j}|^2 \leq \liminf\mathbb{E}|L - \sum\limits_{j=-n}^n \phi_j X_{t-j}|^2 = 0
    \]
    by Fatou's lemma and we are done.
\end{proof}

\begin{proposition}
If $\{X_t\}$ is a stationary process with autocovariance function
$\gamma(\cdot)$ and if $\sum_{j=-\infty}^{\infty}|\psi_j|<\infty$,
then for each $t\in\mathbb{Z}$ the series
\[
\psi(B)X_t=\sum_{j=-\infty}^{\infty}\psi_jX_{t-j}
\]
converges absolutely with probability one and in mean square to the same
limit. If
\[
Y_t=\psi(B)X_t,
\]
then the process $\{Y_t\}$ is stationary with autocovariance function
\[
\gamma_Y(h)=\sum_{j,k=-\infty}^{\infty}
\psi_j\psi_k\gamma(h-j+k).
\]
\end{proposition}

\begin{theorem}
Let $\{X_t\}$ be an ARMA$(p,q)$ process for which the polynomials
$\phi(\cdot)$ and $\theta(\cdot)$ have no common zeroes. Then $\{X_t\}$
is causal if and only if $\phi(z)\neq0$ for all $z\in\mathbb{C}$ such that
$|z|\leq1$. The coefficients $\{\psi_j\}$ in the causal representation
$X_t=\sum_{j=0}^{\infty}\psi_jZ_{t-j}$ are determined by the relation
\[
\psi(z)=\sum_{j=0}^{\infty}\psi_jz^j
=\frac{\theta(z)}{\phi(z)},
\qquad |z|\leq 1.
\]
\end{theorem}
\begin{proof}
Suppose first that $\phi(z)\neq0$ for every $|z|\leq1$. Since the
zero set of the polynomial $\phi$ is finite, there is some $R>1$ such
that $\phi$ has no zero in $|z|<R$. Hence the rational function
\[
\psi(z):=\frac{\theta(z)}{\phi(z)}
\]
is analytic in $|z|<R$ and has a power-series expansion
\[
\psi(z)=\sum_{j=0}^{\infty}\psi_jz^j.
\]
Choose $r$ with $1<r<R$. Cauchy's estimate gives
\[
|\psi_j|\leq M_r r^{-j},
\qquad
M_r:=\max_{|z|=r}|\psi(z)|,
\]
so $\sum_{j=0}^{\infty}|\psi_j|<\infty$. Consequently
\[
Y_t:=\psi(B)Z_t=\sum_{j=0}^{\infty}\psi_jZ_{t-j}
\]
is well defined, and the preceding propositions show that $\{Y_t\}$ is
stationary. Since $\phi(z)\psi(z)=\theta(z)$, absolute summability permits
composition of the filters and gives
\[
\phi(B)Y_t=\theta(B)Z_t=\phi(B)X_t.
\]
Put $U_t=X_t-Y_t$. Then $\{U_t\}$ is stationary and
$\phi(B)U_t=0$. The function $1/\phi(z)$ is also analytic in $|z|<R$
and has absolutely summable Taylor coefficients. Applying the corresponding
filter to the last equality yields
\[
U_t=\frac{1}{\phi}(B)\phi(B)U_t=0.
\]
Thus $X_t=Y_t=\sum_{j=0}^{\infty}\psi_jZ_{t-j}$, so $\{X_t\}$ is
causal.

Conversely, suppose that $\{X_t\}$ is causal, so that
\[
X_t=\psi(B)Z_t=\sum_{j=0}^{\infty}\psi_jZ_{t-j},
\qquad
\sum_{j=0}^{\infty}|\psi_j|<\infty.
\]
Substitution into the ARMA equation gives
\[
\bigl(\phi(B)\psi(B)-\theta(B)\bigr)Z_t=0.
\]
Write
\[
\phi(z)\psi(z)-\theta(z)=\sum_{j=0}^{\infty}a_jz^j.
\]
The sequence $\{a_j\}$ is absolutely summable. Since $\{Z_t\}$ is
white noise with variance $\sigma^2>0$,
\[
0=\operatorname{Var}\left(\sum_{j=0}^{\infty}a_jZ_{t-j}\right)
=\sigma^2\sum_{j=0}^{\infty}|a_j|^2.
\]
Therefore $a_j=0$ for every $j$, and hence
\[
\phi(z)\psi(z)=\theta(z),
\qquad |z|\leq1.
\]
If $\phi(z_0)=0$ for some $|z_0|\leq1$, the absolute convergence of
$\psi(z_0)$ would imply $\theta(z_0)=0$, contradicting the assumption that
$\phi$ and $\theta$ have no common zeroes. Thus $\phi(z)\neq0$ throughout
$|z|\leq1$, and the displayed identity also gives
$\psi(z)=\theta(z)/\phi(z)$. This relation uniquely determines all of the
coefficients $\{\psi_j\}$.
\end{proof}

\begin{definition}[Invertibility]
An ARMA$(p,q)$ process defined by the equations
$\phi(B)X_t=\theta(B)Z_t$ is said to be \emph{invertible} if there exists
a sequence of constants $\{\pi_j\}_{j\geq0}$ such that
\[
\sum_{j=0}^{\infty}|\pi_j|<\infty
\]
and
\[
Z_t=\sum_{j=0}^{\infty}\pi_jX_{t-j},
\qquad t=0,\pm1,\ldots.
\]
\end{definition}

is invertible if and only if $\theta(z)\neq0$ for all $z\in\mathbb{C}$

representation $Z_t=\sum_{j=0}^{\infty}\pi_jX_{t-j}$ are determined by
the relation
\[

=\frac{\phi(z)}{\theta(z)},
\qquad |z|\leq1.
\]
The coefficients $\{\pi_j\}$ can be calculated from recursion relations

\subsection{Stationary Solutions and Laurent Expansions (Section 3.1)}
\begin{theorem}[Stationary solution; Theorem 3.1.3]
If $\phi(z)\ne0$ for $|z|=1$, the equations
$\phi(B)X_t=\theta(B)Z_t$ have a unique stationary solution
\[
X_t=\sum_{j\in\mathbb Z}\psi_jZ_{t-j},\qquad
\frac{\theta(z)}{\phi(z)}=\sum_{j\in\mathbb Z}\psi_jz^j
\quad(r^{-1}<|z|<r)
\]
for some $r>1$. The coefficients are absolutely summable, and the random
series converges absolutely almost surely and in mean square.
Uniqueness is with respect to the given noise sequence, up to almost sure equality.
\end{theorem}
\begin{proposition}[Unit-circle obstruction]
If $\phi$ and $\theta$ have no common zeroes and $\phi$ has a zero on the
unit circle, there is no stationary solution with finite second moments.
Common factors must therefore be removed before applying this criterion.
\end{proposition}
\begin{corollary}[Geometric decay]
For the solution above, there exist $K<\infty$ and $a\in(0,1)$ such that
\[
|\psi_j|\le K a^{|j|},\qquad j\in\mathbb Z.
\]
There also exist $C<\infty$ and $s\in(0,1)$ such that
\[
|\gamma(h)|\le C s^{|h|},\qquad
\sum_{h\in\mathbb Z}|\gamma(h)|<\infty.
\]
In general, a stationary representation is bilateral; causality requires the
negative-index coefficients to vanish.
\end{corollary}

\subsection{Infinite Moving Averages (Section 3.2)}
\begin{definition}[Moving average of infinite order]
An $\mathrm{MA}(\infty)$ of $\{Z_t\}$ is a process of the form
\[
X_t=\sum_{j=0}^{\infty}\psi_jZ_{t-j},\qquad
\sum_{j=0}^{\infty}|\psi_j|<\infty.
\]
The one-sided index set is part of this definition.
\end{definition}
\begin{theorem}[Covariance of an infinite moving average]
The process above is stationary, with mean zero and
\[
\gamma(h)=\sigma^2\sum_{j=0}^{\infty}\psi_j\psi_{j+|h|}.
\]
More generally, $\sum_{j\ge0}\psi_j^2<\infty$ suffices for mean-square
convergence and the same covariance formula, although it does not imply
absolute summability of the coefficients.
\end{theorem}
\begin{proposition}[Finite covariance range; Proposition 3.2.1]
If a zero-mean stationary process satisfies $\gamma(h)=0$ for $|h|>q$
and $\gamma(q)\ne0$, then there is a white noise sequence $\{\varepsilon_t\}$
such that
\[
X_t=\varepsilon_t+\vartheta_1\varepsilon_{t-1}
+\cdots+\vartheta_q\varepsilon_{t-q}.
\]
It can be chosen as the innovation
$\varepsilon_t=X_t-P_{\mathcal M_{t-1}}X_t$, where
$\mathcal M_t=\overline{\operatorname{span}}\{X_s:s\le t\}$.
\end{proposition}
\begin{corollary}[Second-order characterization]
If a zero-mean stationary process has the same autocovariance function as
an ARMA$(p,q)$ process, it too admits an ARMA$(p,q)$ representation with
some white noise sequence. Equality of covariance functions does not imply
equality of finite-dimensional distributions.
\end{corollary}

\subsection{Coefficient and Covariance Identities (Section 3.3)}
For a causal ARMA process, put $\psi_j=0$ for $j<0$,
$\theta_0=1$, and $\theta_j=0$ for $j>q$. The identity
$\phi(z)\psi(z)=\theta(z)$ gives
\[
\psi_0=1,\qquad
\psi_j=\theta_j+\sum_{i=1}^{\min(p,j)}\phi_i\psi_{j-i},\quad j\ge1.
\]
Its covariance function satisfies
\[
\gamma(k)-\sum_{i=1}^{p}\phi_i\gamma(k-i)
=\sigma^2\sum_{j=k}^{q}\theta_j\psi_{j-k},\qquad k\ge0,
\]
where an empty sum is zero and $\gamma(-k)=\gamma(k)$. In particular,
\[
\gamma(k)=\sum_{i=1}^{p}\phi_i\gamma(k-i),\quad k>q.
\]
For an AR$(p)$ process these specialize to the Yule--Walker identities
\[
\gamma(0)-\sum_{i=1}^{p}\phi_i\gamma(i)=\sigma^2,\qquad
\gamma(k)=\sum_{i=1}^{p}\phi_i\gamma(k-i),\quad k\ge1.
\]
These are population identities; no parameter estimation is being assumed.

\subsection{Partial Autocorrelation (Section 3.4)}
\begin{definition}[Partial autocorrelation function]
Let $\gamma(0)>0$ and $\rho(h)=\gamma(h)/\gamma(0)$.
Set $\alpha(1)=\rho(1)$. For $k\ge2$, let
$\mathcal H_k=\overline{\operatorname{span}}\{X_2,\ldots,X_k\}$ and define
\[
\alpha(k)=\operatorname{Corr}\bigl(
X_{k+1}-P_{\mathcal H_k}X_{k+1},\,
X_1-P_{\mathcal H_k}X_1\bigr),
\]
provided the residual variances are positive. For nonzero means, include the
constant $1$ in the projecting space, or first center the process.
\end{definition}
\begin{proposition}[Prediction-coefficient characterization]
Suppose the finite covariance matrices are nonsingular. If
\[
P_{\operatorname{span}\{X_1,\ldots,X_k\}}X_{k+1}
=\sum_{j=1}^{k}\phi_{kj}X_{k+1-j},
\]
then
\[
\alpha(k)=\phi_{kk},\qquad
R_k\boldsymbol\phi_k=\boldsymbol\rho_k,
\quad R_k=[\rho(i-j)]_{i,j=1}^k,
\quad\boldsymbol\rho_k=(\rho(1),\ldots,\rho(k))^\top.
\]
The sample PACF is defined by replacing $\rho$ by the sample ACF in these
equations when the resulting system has a unique solution.
\end{proposition}
\begin{proposition}[Cutoff properties]
For a causal AR$(p)$ process, $\alpha(k)=0$ for $k>p$; if its order is
exactly $p$, then $\alpha(p)=\phi_p\ne0$.
For an MA$(q)$ process, $\gamma(h)=0$ for $|h|>q$.
Thus AR order appears as a PACF cutoff, and MA order as an ACF cutoff.
\end{proposition}

\subsection{Autocovariance Generating Functions (Section 3.5)}
\begin{definition}[Autocovariance generating function]
When the Laurent series converges on an annulus containing the unit circle,
define
\[
G(z)=\sum_{h\in\mathbb Z}\gamma(h)z^h.
\]
For real-valued processes $G(z)=G(z^{-1})$ wherever both sides are defined.
\end{definition}
\begin{proposition}[Filtering identity]
If $X_t=\sum_{j\in\mathbb Z}\psi_jZ_{t-j}$ and
$\psi(z)=\sum_j\psi_jz^j$ converges absolutely on a symmetric annulus
$r^{-1}<|z|<r$, then
\[
G(z)=\sigma^2\psi(z)\psi(z^{-1}).
\]
Consequently, for stationary ARMA equations with no AR zero on the unit circle,
\[
G(z)=\sigma^2\frac{\theta(z)\theta(z^{-1})}
{\phi(z)\phi(z^{-1})}
\]
on a sufficiently small such annulus. A process is white noise precisely
when its covariance generating function is constant.
\end{proposition}
\begin{proposition}[Causal invertible representation; Proposition 3.5.1]
If neither $\phi$ nor $\theta$ has a zero on the unit circle, the same process
admits a representation
\[
\widetilde\phi(B)X_t=\widetilde\theta(B)\widetilde Z_t,
\]
where $\widetilde\phi$ and $\widetilde\theta$ have no zeroes in $|z|\le1$.
The degrees can be retained. This is obtained by reflecting roots inside
the unit circle to their reciprocal conjugates and adjusting the noise variance.
The new noise is white; independence need not be preserved unless Gaussianity
is assumed. Causality and invertibility concern the specified noise representation.
\end{proposition}

\subsection{Linear Difference Equations (Section 3.6)}
\begin{theorem}[General homogeneous solution]
Let $a(z)=1+a_1z+\cdots+a_kz^k$, $a_k\ne0$, have distinct zeroes
$\zeta_1,\ldots,\zeta_r$ of multiplicities $m_1,\ldots,m_r$.
The solution space of $a(B)u_t=0$ has dimension $k$, with basis
\[
\{t^\ell\zeta_i^{-t}:1\le i\le r,\ 0\le\ell<m_i\}.
\]
Thus
\[
u_t=\sum_{i=1}^{r}\sum_{\ell=0}^{m_i-1}c_{i\ell}t^\ell\zeta_i^{-t}.
\]
Any $k$ consecutive initial values determine a unique solution. For real
solutions the coefficients attached to conjugate roots are conjugates.
\end{theorem}
Complex conjugate roots give polynomially modulated sinusoids. If every
$|\zeta_i|>1$, then $|u_t|\le C s^t$ for $t\ge0$ and some $s\in(0,1)$.
The same root structure describes the eventual form of causal ARMA
coefficients and autocovariances.

$\psi(z)=\sum_j\psi_jz^j$ converges absolutely on a symmetric annulus
$r^{-1}<|z|<r$, then
\[
G(z)=\sigma^2\psi(z)\psi(z^{-1}).
\]
Consequently, for stationary ARMA equations with no AR zero on the unit circle,
\[
G(z)=\sigma^2\frac{\theta(z)\theta(z^{-1})}
{\phi(z)\phi(z^{-1})}
\]
on a sufficiently small such annulus. A process is white noise precisely
when its covariance generating function is constant.
\end{proposition}
\begin{proposition}[Causal invertible representation; Proposition 3.5.1]
If neither $\phi$ nor $\theta$ has a zero on the unit circle, the same process
admits a representation
\[
\widetilde\phi(B)X_t=\widetilde\theta(B)\widetilde Z_t,
\]
where $\widetilde\phi$ and $\widetilde\theta$ have no zeroes in $|z|\le1$.
The degrees can be retained. This is obtained by reflecting roots inside
the unit circle to their reciprocal conjugates and adjusting the noise variance.
The new noise is white; independence need not be preserved unless Gaussianity
is assumed. Causality and invertibility concern the specified noise representation.
\end{proposition}

\subsection{Linear Difference Equations (Section 3.6)}
\begin{theorem}[General homogeneous solution]
Let $a(z)=1+a_1z+\cdots+a_kz^k$, $a_k\ne0$, have distinct zeroes
$\zeta_1,\ldots,\zeta_r$ of multiplicities $m_1,\ldots,m_r$.
The solution space of $a(B)u_t=0$ has dimension $k$, with basis
\[
\{t^\ell\zeta_i^{-t}:1\le i\le r,\ 0\le\ell<m_i\}.
\]
Thus
\[
u_t=\sum_{i=1}^{r}\sum_{\ell=0}^{m_i-1}c_{i\ell}t^\ell\zeta_i^{-t}.
\]
Any $k$ consecutive initial values determine a unique solution. For real
solutions the coefficients attached to conjugate roots are conjugates.
\end{theorem}
Complex conjugate roots give polynomially modulated sinusoids. If every
$|\zeta_i|>1$, then $|u_t|\le C s^t$ for $t\ge0$ and some $s\in(0,1)$.
The same root structure describes the eventual form of causal ARMA
coefficients and autocovariances.
